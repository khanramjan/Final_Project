% Section 1: Introduction
\section{Introduction}
\begin{frame}
    \frametitle{Introduction}
    \begin{itemize}
        \item \textbf{Background:} Modern universities generate significant amounts of resources that can be repurposed, but lack an organized system to facilitate donations.
        \item \textbf{What is the DMS?} The Campus Based Donation Management System (DMS) is a centralized, digital platform connecting donors (students, faculty) with recipients in need.
        \item \textbf{Why it matters:} Promotes a culture of sustainability, resource-sharing, and community support within the campus ecosystem.
    \end{itemize}
\end{frame}

% Section 2: Problem Statement
\section{Problem Statement}
\begin{frame}
    \frametitle{Problem Statement}
    \begin{block}{Current Challenges in Campus Resource Sharing}
        \begin{itemize}
            \item \textbf{Lack of Coordination:} No centralized platform for students/faculty to donate usable goods (books, electronics, clothing).
            \item \textbf{Inefficiency:} Information relies on fragmented social media groups or word of mouth.
            \item \textbf{Transparency Issues:} Donors often do not know if their items reached those genuinely in need.
            \item \textbf{Tracking:} Administrators struggle to track campaigns and overall impact.
        \end{itemize}
    \end{block}
\end{frame}

% Section 3: Objectives
\section{Objectives}
\begin{frame}
    \frametitle{Objectives}
    The primary objectives of the Campus Based Donation Management System are:
    \vspace{0.3cm}
    \begin{enumerate}
        \item \textbf{Centralized Platform:} Develop a robust web application for managing campus-wide donations seamlessly.
        \item \textbf{Campaign Management:} Allow administrators to run targeted donation campaigns (e.g., winter clothing drives, book banks).
        \item \textbf{Transparency \& Tracking:} Provide real-time tracking of donations from pledge to delivery.
        \item \textbf{User-Friendly Experience:} Create a highly responsive, accessible, and intuitive UI for both donors and admins.
        \item \textbf{Analytics:} Deliver insight via a dashboard to measure community impact and engagement.
    \end{enumerate}
\end{frame}

% Section 4: Architecture & Methodology
\section{System Architecture \& Technologies}
\begin{frame}
    \frametitle{System Architecture \& Tech Stack}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Frontend (Client-Side)}
        \begin{itemize}
            \item React with TypeScript
            \item Redux Toolkit for State Management
            \item Tailwind CSS for UI Design
        \end{itemize}
        
        \vspace{0.4cm}
        \textbf{Backend (Server-Side)}
        \begin{itemize}
            \item .NET Core Web API
            \item Entity Framework Core
            \item RESTful API Architecture
        \end{itemize}
        
        \column{0.5\textwidth}
        \begin{block}{Architectural Highlights}
            \begin{itemize}
                \item Role-Based Access Control (Admin vs. Donor)
                \item JWT-based Authentication
                \item Responsive Design implementation for mobile and desktop
                \item Separation of concerns (Frontend / API / Database)
            \end{itemize}
        \end{block}
    \end{columns}
\end{frame}

% Section 5: Key Features
\section{Key Features}
\begin{frame}
    \frametitle{Key Features of the System}
    \begin{description}
        \item[Donor Management] Secure registration, profile management, and donation history tracking for users.
        \item[Item Posting \& Claiming] Donors can list items; verified users can request/claim them.
        \item[Campaign Dashboard] Dedicated admin interface to monitor active campaigns, progress bars, and total impact.
        \item[Analytics \& Reporting] Visual dashboards illustrating donation trends across the university.
    \end{description}
